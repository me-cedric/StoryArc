const stages = ['draft', 'review', 'approved'];
 
function advance(current) {
const i = stages.indexOf(current);
return stages[Math.min(i + 1, stages.length - 1)];
}
 
let state = 'draft';
state = advance(state); // 'review'
state = advance(state); // 'approved'
